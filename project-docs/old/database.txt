Table risk_assessments {
  risk_assessment_id uuid [pk, default: `gen_random_uuid()`]
  risk_code varchar(50) [not null, unique]
  asset_id uuid
  business_process_id uuid
  title varchar(255) [not null]
  description text
  likelihood smallint [not null]
  impact smallint [not null]
  risk_score smallint [not null, note: 'likelihood x impact']
  risk_level varchar(20) [not null]
  residual_likelihood smallint
  residual_impact smallint
  residual_score smallint
  status varchar(30) [not null, default: 'draft']
  assessed_by_user_id uuid
  previous_risk_assessment_id uuid
  assessed_at timestamptz [not null, default: `now()`]
  closed_at timestamptz
  created_at timestamptz [not null, default: `now()`]
  updated_at timestamptz [not null, default: `now()`]

  indexes {
    (asset_id)
    (business_process_id)
    (status, risk_level)
    (assessed_by_user_id)
  }

  Note: '''
  CHECK (likelihood BETWEEN 1 AND 5)

  CHECK (impact BETWEEN 1 AND 5)

  CHECK (risk_score = likelihood * impact)

  CHECK (risk_level IN ('low', 'medium', 'high', 'critical'))

  CHECK (
    residual_likelihood IS NULL
    OR residual_likelihood BETWEEN 1 AND 5
  )

  CHECK (
    residual_impact IS NULL
    OR residual_impact BETWEEN 1 AND 5
  )

  CHECK (
    residual_score IS NULL
    OR (
      residual_likelihood IS NOT NULL
      AND residual_impact IS NOT NULL
      AND residual_score = residual_likelihood * residual_impact
    )
  )

  CHECK (
    status IN (
      'draft',
      'pending_approval',
      'approved',
      'in_treatment',
      'closed',
      'rejected'
    )
  )

  CHECK (
    (asset_id IS NOT NULL AND business_process_id IS NULL)
    OR (asset_id IS NULL AND business_process_id IS NOT NULL)
  )
  '''
}

// =========================================================
// V3 DESIGN DECISIONS
// =========================================================
// 1. AI uses pre-trained models only. No training, retraining, or fine-tuning is in scope.
// 2. AI flow: Integration -> Log Source -> Security Event -> Pre-trained AI Model -> AI Alert -> Explanation/Feedback.
// 3. Security events provide traceability for external_event_id and AI alerts.
// 4. Risk assessment targets exactly one of Asset or Business Process.
// 5. Incident assignment targets exactly one of User or Team.
// 6. Training campaign target targets exactly one of User or Department.
// 7. Controlled values use CHECK constraints rather than PostgreSQL ENUM/lookup tables.
// 8. Risk scores are stored as historical snapshots and protected by CHECK constraints.
// 9. Frequently queried foreign keys and filter/time fields are indexed.
// 10. Partial unique index for one active model per model family is required in PostgreSQL.
